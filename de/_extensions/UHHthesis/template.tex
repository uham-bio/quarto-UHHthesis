% "Appendices"-Gruppenheader aus dem Inhaltsverzeichnis unterdruecken.
% Quarto erzeugt automatisch \addcontentsline{toc}{part}{Anhänge};
% da wir article-Klasse ohne echte Parts verwenden, ueberspringen
% wir alle part-Ebenen-Eintraege im ToC.
\makeatletter
\let\orig@addcontentsline\addcontentsline
\renewcommand{\addcontentsline}[3]{%
  \def\@tempa{part}\def\@tempb{#2}%
  \ifx\@tempa\@tempb\else
    \orig@addcontentsline{#1}{#2}{#3}%
  \fi}
\makeatother
% }}}

  \setlength{\itemindent}{0pt}
  \setlength{\leftmargin}{0pt}
  \setlength{\parsep}{0pt}
  % turn on hanging indent if param 1 is 1
  \ifodd #1
   \setlength{\leftmargin}{\cslhangindent}
   \setlength{\itemindent}{-1\cslhangindent}
  \fi
  % set entry spacing
  \setlength{\itemsep}{#2\baselineskip}}}
 {\end{list}}
